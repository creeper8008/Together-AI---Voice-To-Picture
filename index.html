async function generateImage(prompt) { // 1. トークンのクリーニング const token = hfInput.value.trim().replace(/[^\x21-\x7E]/g, ""); // 2. モデルの選定 (現在、無料枠で最も安定しているものに変更) const MODEL_ID = "runwayml/stable-diffusion-v1-5"; loader.style.display = 'block'; imageImg.style.display = 'none'; addLog(`Generating with ${MODEL_ID}...`); try { const response = await fetch( `https://api-inference.huggingface.co/models/${MODEL_ID}`, { method: "POST", headers: { "Authorization": `Bearer ${token}`, "Content-Type": "application/json", // 余計なヘッダーを削除してCORSエラーを防止 }, body: JSON.stringify({ inputs: prompt, options: { wait_for_model: true } }), } ); // 404や401などのエラーを詳細にキャッチ if (!response.ok) { const errorText = await response.text(); if (response.status === 404) { throw new Error("Model not found. Try changing the MODEL_ID."); } throw new Error(`API Error ${response.status}: ${errorText}`); } const blob = await response.blob(); if (blob.size < 1000) { // 極端に小さいデータは画像ではない throw new Error("Generated data is too small (possible API error)"); } const imageUrl = URL.createObjectURL(blob); imageImg.src = imageUrl; imageImg.style.display = 'block'; addLog("Success!"); } catch (e) { addLog(`Error: ${e.message}`); console.error("Detailed Error:", e); // --- 最終手段のフォールバック --- if (e.message.includes("fetch") || e.message.includes("404")) { addLog("Switching to fallback generator..."); fallbackPollinations(prompt); } } finally { loader.style.display = 'none'; } } // Hugging Faceが全滅した時のための予備（CORSの影響を受けない方法） function fallbackPollinations(prompt) { const seed = Math.floor(Math.random() * 100000); const url = `https://image.pollinations.ai/prompt/${encodeURIComponent(prompt)}?seed=${seed}&nologo=true`; imageImg.src = url; imageImg.style.display = 'block'; addLog("Fallback: Generated via Pollinations"); }
